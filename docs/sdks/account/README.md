### Standalone function

The standalone function version of this method:

```typescript
import { ModerationApiCore } from "@moderation-api/sdk/core.js";
import { accountRetrieve } from "@moderation-api/sdk/funcs/account-retrieve.js";

// Use `ModerationApiCore` for best tree-shaking performance.
// You can create one instance of it to use across an application.
const moderationApi = new ModerationApiCore({
  secretKey: process.env["MODAPI_SECRET_KEY"] ?? "",
});

async function run() {
  const res = await accountRetrieve(moderationApi);
  if (res.ok) {
    const { value: result } = res;
    console.log(result);
  } else {
    console.log("accountRetrieve failed:", res.error);
  }
}

run();
```
